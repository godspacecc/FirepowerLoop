using System;using System.IO;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Game;
public class Script
{
static readonly List<ResourceLoadHandle> handles=new();static readonly List<string> report=new();static int completed;static bool ended;static double started;
static string file="E:/GitHub/FirepowerLoop/Backup/Artifacts/SupportItems-20261102/airstrike/runtime-load.txt";
public static object Main()
{
if(!Application.isPlaying||Application.dataPath!="E:/GitHub/FirepowerLoop/Client/Assets"||!(ResourceLoader.Current is YooAssetResourceLoader))throw new Exception("需要正常Play中的真实YooAsset加载器");
if(File.Exists(file))File.Delete(file);started=EditorApplication.timeSinceStartup;report.Add("loader="+ResourceLoader.Current.GetType().FullName);report.Add("mode="+YooAsset.YooAssets.GetPackage("DefaultPackage").PackageName);
report.Add("playMode="+typeof(YooAsset.ResourcePackage).GetField("_playMode",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic).GetValue(YooAsset.YooAssets.GetPackage("DefaultPackage")));
foreach(string address in new[]{"Prefabs/SceneObjects/Arcade/ArcadeObstacleVisualSet","Prefabs/UI/Survivor/SurvivorHud"})
{
var handle=ResourceLoader.LoadAssetAsync<GameObject>(address,asset=>
{
if(ended)return;
try{if(asset==null)throw new Exception("asset null");
if(address.Contains("Obstacle")){
var host=new GameObject("MissileRuntimeCheck");var view=new ArcadeCombatViewController();var flags=System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance;
try{typeof(ArcadeCombatViewController).GetField("root",flags).SetValue(view,host);
typeof(ArcadeCombatViewController).GetMethod("BindMissileTemplate",flags).Invoke(view,new object[]{asset.transform.Find("Content")});
var point=new Vector2(2,3);view.SetMissileStrike(point,1.6f,0,true);var instance=host.transform.GetChild(0);var body=instance.Find("Body");var ring=instance.Find("Warning");if(body.gameObject.activeSelf||ring.localScale.x!=1.6f||instance.localPosition!=new Vector3(2,0,3))throw new Exception("预警位置或半径错误");
view.SetMissileStrike(point,1.6f,1.10f,true);float height=body.localPosition.y;view.SetMissileStrike(point,1.6f,1.35f,true);if(!body.gameObject.activeSelf||body.localPosition.y>=height)throw new Exception("导弹没有垂直下落");
view.SetMissileStrike(point,1.6f,1.65f,false);if(instance.gameObject.activeSelf)throw new Exception("爆炸后未回收");view.SetMissileStrike(Vector2.zero,2.6f,0,true);if(host.transform.childCount!=1)throw new Exception("未复用实例");
foreach(var r in instance.GetComponentsInChildren<Renderer>(true)){if(r.sharedMaterial==null||!r.sharedMaterial.shader.isSupported)throw new Exception("材质无效");}
}finally{typeof(ArcadeCombatViewController).GetMethod("ClearMissileVisual",flags).Invoke(view,null);UnityEngine.Object.Destroy(host);}
}else{var hud=asset.GetComponent<SurvivorHudUI>();var slot=UnityEngine.Object.Instantiate(hud.WeaponSlots.itemPrefab).GetComponent<SurvivorWeaponSlotBoxUI>();
try{var model=new SurvivorModel();var flags=System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Instance;typeof(SurvivorModel).GetField("RocketClock",flags).SetValue(model,2.5f);typeof(SurvivorModel).GetField("RocketInterval",flags).SetValue(model,5f);
typeof(SurvivorHudUICtrl).GetMethod("RenderWeaponSlot",System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Static).Invoke(null,new object[]{slot,model,"RocketSupport"});
if(slot.MissileIcon==null||slot.Icon.sprite!=slot.MissileIcon||!slot.AmmoRing.gameObject.activeSelf||Mathf.Abs(slot.AmmoRing.fillAmount-.5f)>.001f)throw new Exception("导弹图标或冷却圈无效");
}finally{UnityEngine.Object.Destroy(slot.gameObject);}}
report.Add(address+" : PASS actual YooAsset prefab and serialized dependencies");}catch(Exception ex){report.Add(address+" : FAIL "+ex.Message);}completed++;
});handles.Add(handle);
}
EditorApplication.update+=Poll;return "2个异步真实加载已发起";
}
static void Poll()
{
if(completed<2&&EditorApplication.timeSinceStartup-started<30)return;
ended=true;EditorApplication.update-=Poll;
foreach(var handle in handles){ResourceLoader.Release(handle);report.Add("released="+(!(handle.RawHandle is YooAsset.AssetHandle raw)||!raw.IsValid));}
report.Add("completed="+completed+"; failed="+report.Exists(s=>s.Contains("FAIL")));File.WriteAllLines(file,report);
}
}
