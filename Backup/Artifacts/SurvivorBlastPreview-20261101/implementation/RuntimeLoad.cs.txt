using System;using System.IO;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Game;
public class Script
{
static readonly List<ResourceLoadHandle> handles=new();static readonly List<string> report=new();static int completed;static bool ended;static double started;
static string file="E:/GitHub/FirepowerLoop/Backup/Artifacts/SurvivorBlastPreview-20261101/implementation/runtime-load.txt";
public static object Main()
{
if(!Application.isPlaying||Application.dataPath!="E:/GitHub/FirepowerLoop/Client/Assets"||!(ResourceLoader.Current is YooAssetResourceLoader))throw new Exception("需要正常Play中的真实YooAsset加载器");
if(File.Exists(file))File.Delete(file);started=EditorApplication.timeSinceStartup;report.Add("loader="+ResourceLoader.Current.GetType().FullName);report.Add("mode="+YooAsset.YooAssets.GetPackage("DefaultPackage").PackageName);
report.Add("playMode="+typeof(YooAsset.ResourcePackage).GetField("_playMode",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic).GetValue(YooAsset.YooAssets.GetPackage("DefaultPackage")));
foreach(string name in new[]{"GrayboxBullet"})
{
string address="Prefabs/Combat/"+name;var handle=ResourceLoader.LoadAssetAsync<GameObject>(address,asset=>
{
if(ended)return;
try{if(asset==null)throw new Exception("asset null");var projectile=asset.GetComponent<ProjectileView>();var blast=projectile!=null?projectile.SurvivorBlastTemplate:null;if(blast==null||!blast.Validate(out var error))throw new Exception("short blast dependency invalid");foreach(var r in blast.GetComponentsInChildren<ParticleSystemRenderer>(true)){var mat=r.sharedMaterial;if(mat==null||mat.shader==null||!mat.shader.isSupported)throw new Exception("material shader invalid");foreach(var property in mat.GetTexturePropertyNames())if(property=="_BaseMap"&&mat.GetTexture(property)==null)throw new Exception("base texture invalid");}report.Add(address+" : PASS real projectile, original short-blast reference, 3 particle systems/materials/textures/shader and manual clock settings");}catch(Exception ex){report.Add(address+" : FAIL "+ex.Message);}completed++;
});handles.Add(handle);
}
EditorApplication.update+=Poll;return "1个异步真实UI加载已发起";
}
static void Poll()
{
if(completed<1&&EditorApplication.timeSinceStartup-started<30)return;
ended=true;EditorApplication.update-=Poll;
foreach(var handle in handles){ResourceLoader.Release(handle);report.Add("released="+(!(handle.RawHandle is YooAsset.AssetHandle raw)||!raw.IsValid));}
report.Add("completed="+completed+"; failed="+report.Exists(s=>s.Contains("FAIL")));File.WriteAllLines(file,report);
}
}
