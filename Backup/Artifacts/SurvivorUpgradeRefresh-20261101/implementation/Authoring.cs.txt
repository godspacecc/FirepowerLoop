static SurvivorUpgradeShape Shape(Transform parent,string name,float x,float y,float w,float h,Color color,Vector2[] points)

static Button Button(Transform parent,string value,float x,float y,float w,float h,bool primary,bool hovered=false)
{
var shadow=Panel(parent,"ButtonShadow",x,y-5,w,h,new Color(0,0,0,.4f),5);

// 图标保持真实按钮底板，计数置于刷新箭头右侧；临时预览不写正式资源。
var ink=primary?new Color(.14f,.18f,.18f):cream;
if(primary){Stroke(edge.transform,-14,0,-4,-10,4.5f,ink);Stroke(edge.transform,-4,-10,16,12,4.5f,ink);}
else
{

}

return button;
}
static void Stroke(Transform parent,float x1,float y1,float x2,float y2,float width,Color color)
{var a=new Vector2(x1,y1);var b=new Vector2(x2,y2);var line=Panel(parent,"IconStroke",(x1+x2)*.5f,(y1+y2)*.5f,Vector2.Distance(a,b)+1,width,color,0);line.rectTransform.localRotation=Quaternion.Euler(0,0,Mathf.Atan2(y2-y1,x2-x1)*Mathf.Rad2Deg);}
