using System;using System.IO;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Game;
public class Script
{
static readonly List<ResourceLoadHandle> handles=new();static readonly List<string> report=new();static int completed;static bool ended;static double started;
static string file="E:/GitHub/FirepowerLoop/Backup/Artifacts/HitFeedbackReference-20261102/implementation/runtime-load.txt";
public static object Main()
{
if(!Application.isPlaying||Application.dataPath!="E:/GitHub/FirepowerLoop/Client/Assets"||!(ResourceLoader.Current is YooAssetResourceLoader))throw new Exception("需要正常Play中的真实YooAsset加载器");
if(File.Exists(file))File.Delete(file);started=EditorApplication.timeSinceStartup;report.Add("loader="+ResourceLoader.Current.GetType().FullName);report.Add("mode="+YooAsset.YooAssets.GetPackage("DefaultPackage").PackageName);
report.Add("playMode="+typeof(YooAsset.ResourcePackage).GetField("_playMode",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic).GetValue(YooAsset.YooAssets.GetPackage("DefaultPackage")));
foreach(string address in new[]{"Prefabs/Characters/GrayboxPlayer","Prefabs/Characters/ArcadeZombieShambler","Prefabs/Characters/ArcadeZombieGaunt","Prefabs/Characters/ArcadeZombieBloated","Prefabs/Characters/ArcadeZombieExploder","Prefabs/Characters/ArcadeZombieCaster","Prefabs/Characters/ArcadeZombieShield","Prefabs/Characters/ArcadeZombieEnchantHealth","Prefabs/Characters/ArcadeZombieEnchantSpeed","Prefabs/Characters/ArcadeZombieEnchantShield","Prefabs/UI/Survivor/SurvivorHud"})
{
var handle=ResourceLoader.LoadAssetAsync<GameObject>(address,asset=>
{
if(ended)return;
try{if(asset==null)throw new Exception("asset null");
if(!address.Contains("/UI/")){
var feedback=asset.GetComponentInChildren<SurvivorHitFeedbackView>(true);if(feedback==null||!feedback.Validate(out var error))throw new Exception("feedback invalid");
foreach(var renderer in feedback.GetComponentsInChildren<Renderer>(true))foreach(var mat in renderer.sharedMaterials)if(mat==null||mat.shader==null||!mat.shader.isSupported)throw new Exception("material/shader invalid");
if(address.EndsWith("GrayboxPlayer")||address.EndsWith("ArcadeZombieShambler")){
var holder=new GameObject("TEMP_RuntimeHitLifecycle");holder.SetActive(false);try{var actor=UnityEngine.Object.Instantiate(asset,holder.transform);foreach(var behaviour in actor.GetComponentsInChildren<MonoBehaviour>(true))if(!(behaviour is SurvivorHitFeedbackView))behaviour.enabled=false;foreach(var collider in actor.GetComponentsInChildren<Collider>(true))collider.enabled=false;var f=actor.GetComponentInChildren<SurvivorHitFeedbackView>(true);var animator=address.EndsWith("GrayboxPlayer")?actor.GetComponent<PlayerView>().Animator:actor.GetComponent<EnemyView>().Animator;f.Bind(animator,address.EndsWith("GrayboxPlayer"));holder.SetActive(true);f.Play(Vector3.right);f.Tick(.04f);f.ApplyPose();holder.SetActive(false);foreach(var renderer in f.GetComponentsInChildren<Renderer>(true))if(renderer.gameObject.activeSelf)throw new Exception("real OnDisable left transient enabled");report.Add(address+" : real Play activation/OnDisable cleanup PASS");}finally{UnityEngine.Object.Destroy(holder);}}
}else{var hud=asset.GetComponent<SurvivorHudUI>();if(hud.HurtEdges==null||hud.HurtEdges.GetComponentsInChildren<UnityEngine.UI.Image>(true).Length!=4)throw new Exception("HUD binding missing");}
report.Add(address+" : PASS actual YooAsset prefab and serialized dependencies");}catch(Exception ex){report.Add(address+" : FAIL "+ex.Message);}completed++;
});handles.Add(handle);
}
EditorApplication.update+=Poll;return "11个异步真实加载已发起";
}
static void Poll()
{
if(completed<11&&EditorApplication.timeSinceStartup-started<30)return;
ended=true;EditorApplication.update-=Poll;
foreach(var handle in handles){ResourceLoader.Release(handle);report.Add("released="+(!(handle.RawHandle is YooAsset.AssetHandle raw)||!raw.IsValid));}
report.Add("completed="+completed+"; failed="+report.Exists(s=>s.Contains("FAIL")));File.WriteAllLines(file,report);
}
}
