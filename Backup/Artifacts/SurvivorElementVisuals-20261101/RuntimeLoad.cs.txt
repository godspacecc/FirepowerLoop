using System;using System.IO;using System.Collections.Generic;using UnityEngine;using UnityEditor;using Game;
public class Script
{
static readonly List<ResourceLoadHandle> handles=new();static readonly List<string> report=new();static int completed;static bool ended;static double started;
static string file="E:/GitHub/FirepowerLoop/Backup/Artifacts/SurvivorElementVisuals-20261101/runtime-load.txt";
public static object Main()
{
if(!Application.isPlaying||Application.dataPath!="E:/GitHub/FirepowerLoop/Client/Assets"||!(ResourceLoader.Current is YooAssetResourceLoader))throw new Exception("需要正常Play中的真实YooAsset加载器");
if(File.Exists(file))File.Delete(file);started=EditorApplication.timeSinceStartup;report.Add("loader="+ResourceLoader.Current.GetType().FullName);report.Add("mode="+YooAsset.YooAssets.GetPackage("DefaultPackage").PackageName);
report.Add("playMode="+typeof(YooAsset.ResourcePackage).GetField("_playMode",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic).GetValue(YooAsset.YooAssets.GetPackage("DefaultPackage")));
foreach(string name in new[]{"ArcadeZombieShambler","ArcadeZombieGaunt","ArcadeZombieBloated"})
{
string address="Prefabs/Characters/"+name;var handle=ResourceLoader.LoadAssetAsync<GameObject>(address,asset=>
{
if(ended)return;
try{if(asset==null)throw new Exception("asset null");var feedback=asset.GetComponentInChildren<EnemyElementFeedback>(true);if(feedback==null||!feedback.Validate(out var error))throw new Exception("missing or invalid element dependencies");if(feedback.transform.parent!=asset.GetComponent<EnemyView>().Animator.transform.parent)throw new Exception("body attachment invalid");foreach(var ps in feedback.GetComponentsInChildren<ParticleSystem>(true)){var material=ps.GetComponent<ParticleSystemRenderer>().sharedMaterial;if(material==null||!material.shader.isSupported||material.GetTexture("_BaseMap")==null)throw new Exception("material shader texture invalid");}report.Add(address+" : PASS prefab, serialized particles/materials/textures/shader, body references");}catch(Exception ex){report.Add(address+" : FAIL "+ex.Message);}completed++;
});handles.Add(handle);
}
EditorApplication.update+=Poll;return "3个异步真实加载已发起";
}
static void Poll()
{
if(completed<3&&EditorApplication.timeSinceStartup-started<30)return;
ended=true;EditorApplication.update-=Poll;
foreach(var handle in handles){ResourceLoader.Release(handle);report.Add("released="+(!(handle.RawHandle is YooAsset.AssetHandle raw)||!raw.IsValid));}
report.Add("completed="+completed+"; failed="+report.Exists(s=>s.Contains("FAIL")));File.WriteAllLines(file,report);
}
}
