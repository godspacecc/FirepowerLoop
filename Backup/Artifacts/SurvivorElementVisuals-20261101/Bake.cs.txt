using System;using System.Linq;using System.IO;using UnityEngine;using UnityEditor;using Object=UnityEngine.Object;
public class Script
{
static string art="Assets/Game/Art/Survivor/Elements";
static void Folder(string path){if(AssetDatabase.IsValidFolder(path))return;var split=path.LastIndexOf('/');Folder(path.Substring(0,split));AssetDatabase.CreateFolder(path.Substring(0,split),path.Substring(split+1));}
static Material Material(bool fire)
{
string stem=fire?"ElementFire":"ElementFrost";string png=art+"/"+stem+"Soft.png";
if(!File.Exists(png))
{
var tex=new Texture2D(32,32,TextureFormat.RGBA32,false);for(int y=0;y<32;y++)for(int x=0;x<32;x++){float radius=Vector2.Distance(new Vector2(x,y),new Vector2(15.5f,15.5f))/15.5f;tex.SetPixel(x,y,new Color(1,1,1,Mathf.Pow(Mathf.Clamp01(1-radius),fire?1.3f:.6f)));}tex.Apply();File.WriteAllBytes(png,tex.EncodeToPNG());Object.DestroyImmediate(tex);AssetDatabase.ImportAsset(png,ImportAssetOptions.ForceSynchronousImport);
var importer=(TextureImporter)AssetImporter.GetAtPath(png);importer.alphaSource=TextureImporterAlphaSource.FromInput;importer.alphaIsTransparency=true;importer.mipmapEnabled=false;importer.wrapMode=TextureWrapMode.Clamp;importer.filterMode=FilterMode.Bilinear;importer.SaveAndReimport();
}
string path=art+"/"+stem+".mat";var material=AssetDatabase.LoadAssetAtPath<Material>(path);if(material==null){material=new Material(Shader.Find("Universal Render Pipeline/Particles/Unlit"));AssetDatabase.CreateAsset(material,path);}
material.SetTexture("_BaseMap",AssetDatabase.LoadAssetAtPath<Texture2D>(png));material.SetTexture("_MainTex",AssetDatabase.LoadAssetAtPath<Texture2D>(png));material.SetFloat("_Surface",1);material.SetFloat("_Blend",0);material.SetFloat("_SrcBlend",5);material.SetFloat("_DstBlend",10);material.SetFloat("_ZWrite",0);material.EnableKeyword("_SURFACE_TYPE_TRANSPARENT");material.renderQueue=3000;EditorUtility.SetDirty(material);AssetDatabase.SaveAssetIfDirty(material);return material;
}
static ParticleSystem Aura(Transform parent,bool fire)
{
var go=new GameObject(fire?"ElementFire":"ElementFrost");go.transform.SetParent(parent,false);go.transform.localPosition=new Vector3(0,fire?.9f:.3f,0);go.transform.localRotation=Quaternion.Euler(-90,0,0);
var ps=go.AddComponent<ParticleSystem>();ps.Stop(true,ParticleSystemStopBehavior.StopEmittingAndClear);var main=ps.main;main.loop=true;main.duration=2;main.playOnAwake=false;main.startLifetime=fire?.85f:1;main.startSpeed=fire?1.1f:.16f;main.startSize=fire?.3f:.08f;main.maxParticles=80;main.simulationSpace=ParticleSystemSimulationSpace.Local;main.startColor=fire?Color.white:new Color(.45f,.9f,1,.9f);if(fire){main.startSize3D=true;main.startSizeX=.28f;main.startSizeY=.55f;main.startSizeZ=.28f;}ps.useAutoRandomSeed=false;ps.randomSeed=37;
var emission=ps.emission;emission.rateOverTime=fire?40:18;var shape=ps.shape;shape.shapeType=ParticleSystemShapeType.Cone;shape.angle=fire?12:30;shape.radius=fire?.25f:.4f;
var color=ps.colorOverLifetime;color.enabled=true;var gradient=new Gradient();gradient.SetKeys(new[]{new GradientColorKey(fire?new Color(1,.9f,.18f):new Color(.7f,1,1),0),new GradientColorKey(fire?new Color(1,.22f,.015f):new Color(.2f,.65f,1),.6f)},new[]{new GradientAlphaKey(0,0),new GradientAlphaKey(.85f,.15f),new GradientAlphaKey(0,1)});color.color=gradient;var size=ps.sizeOverLifetime;size.enabled=true;size.size=new ParticleSystem.MinMaxCurve(1,AnimationCurve.EaseInOut(0,1,1,.1f));go.GetComponent<ParticleSystemRenderer>().sharedMaterial=Material(fire);return ps;
}
public static object Main()
{
if(Application.dataPath!="E:/GitHub/FirepowerLoop/Client/Assets"||EditorApplication.isPlaying||EditorApplication.isCompiling)throw new Exception("需在本项目空闲编辑状态制作");Folder(art);string prefab="Assets/Game/Prefabs/Combat/SurvivorEnemyElements.prefab";var root=new GameObject("SurvivorEnemyElements");
try{var fire=Aura(root.transform,true);var frost=Aura(root.transform,false);var feedback=root.AddComponent<Game.EnemyElementFeedback>();var so=new SerializedObject(feedback);so.FindProperty("fireParticles").objectReferenceValue=fire;so.FindProperty("frostParticles").objectReferenceValue=frost;so.ApplyModifiedPropertiesWithoutUndo();PrefabUtility.SaveAsPrefabAsset(root,prefab);}finally{Object.DestroyImmediate(root);}
foreach(var name in new[]{"ArcadeZombieShambler","ArcadeZombieGaunt","ArcadeZombieBloated"})
{
string path="Assets/Game/Prefabs/Characters/"+name+".prefab";var enemy=PrefabUtility.LoadPrefabContents(path);try
{
var element=enemy.GetComponentInChildren<Game.EnemyElementFeedback>(true);if(element==null){var instance=(GameObject)PrefabUtility.InstantiatePrefab(AssetDatabase.LoadAssetAtPath<GameObject>(prefab),enemy.transform);instance.transform.SetLocalPositionAndRotation(Vector3.zero,Quaternion.identity);element=instance.GetComponent<Game.EnemyElementFeedback>();}
// 采用世界姿态不变的挂接，使站立大小保持预览，而爬出时随身体俯仰和升降。
var body=enemy.GetComponent<Game.EnemyView>().Animator.transform.parent;if(element.transform.parent!=body){element.transform.SetParent(body,true);PrefabUtility.RecordPrefabInstancePropertyModifications(element.transform);}
var renderers=enemy.GetComponentsInChildren<SkinnedMeshRenderer>(true);var serialized=new SerializedObject(element);var list=serialized.FindProperty("frostRenderers");list.arraySize=renderers.Length;for(int i=0;i<renderers.Length;i++)list.GetArrayElementAtIndex(i).objectReferenceValue=renderers[i];serialized.ApplyModifiedPropertiesWithoutUndo();PrefabUtility.RecordPrefabInstancePropertyModifications(element);PrefabUtility.SaveAsPrefabAsset(enemy,path);
}finally{PrefabUtility.UnloadPrefabContents(enemy);}
}
return "采用版元素依赖已保存：3个正式丧尸，共用Prefab、2材质、2纹理；无动态新地址";
}
}
