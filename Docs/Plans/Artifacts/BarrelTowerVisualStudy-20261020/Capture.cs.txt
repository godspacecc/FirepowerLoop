using System;
using System.IO;
using System.Linq;
using System.Reflection;
using Game;
using Luban;
using FirepowerLoop.Arcade;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;
using Object=UnityEngine.Object;

/// <summary>只在临时隔离场景采样正式表现；不修改资源、规则、存档或正式场景。</summary>
public static class BarrelTowerVisualStudy
{
    public static string Run()
    {
        if(Application.isPlaying||Application.dataPath.Replace('\\','/')!="E:/GitHub/FirepowerLoop/Client/Assets")throw new Exception("预览项目门禁不匹配");
        var prior=SceneManager.GetActiveScene();var oldTarget=RenderTexture.active;
        var scene=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Additive);
        var ambient=RenderSettings.ambientLight;var mode=RenderSettings.ambientMode;
        RenderTexture target=null;Texture2D image=null;ArcadeCombatViewController view=null;
        try
        {
            SceneManager.SetActiveScene(scene);RenderSettings.ambientMode=UnityEngine.Rendering.AmbientMode.Flat;RenderSettings.ambientLight=new Color(.65f,.67f,.7f);
            var root=new GameObject("FormalVisualStudy");view=new ArcadeCombatViewController();
            var flags=BindingFlags.Instance|BindingFlags.NonPublic;
            typeof(ArcadeCombatViewController).GetField("root",flags).SetValue(view,root);
            var templates=AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Game/Prefabs/SceneObjects/Arcade/ArcadeObstacleVisualSet.prefab").transform.Find("Content");
            typeof(ArcadeCombatViewController).GetField("contentTemplates",flags).SetValue(view,templates);
            var tables=new Tables(n=>new ByteBuf(File.ReadAllBytes(Path.Combine(Application.dataPath,"Game/Config",n+".bytes"))));
            var provider=new LubanRuntimeConfigProvider(()=>tables);
            if(!provider.TryGetArcadeCatalog(out var catalog,out var error))throw new Exception(error);
            view.ConfigureRoundBounds(19.5f,19.5f,.55f);
            view.SetRoundLayout(ArcadeRoundLayoutParameters.FromConfig(tables.TbArcadeRoundLayout.DataList.First(p=>p.ArenaConfigNumber==3&&p.RoundNumber==1)));
            ((GameObject)typeof(ArcadeCombatViewController).GetField("roundEnvironmentRoot",flags).GetValue(view)).SetActive(false);
            view.RecordRoundEnemyPosition(7,new Vector2(-1.5f,0));view.SetEnemyContent(7,1,0,false);view.SetRoundEnemyHealth(7,100,100);
            var player=Object.Instantiate(AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Game/Prefabs/Characters/GrayboxPlayer.prefab"));player.SetActive(true);
            player.transform.position=new Vector3(-4.4f,0,0);player.transform.rotation=Quaternion.Euler(0,160,0);
            var presenter=player.GetComponent<CharacterWeaponPresenter>();
            if(!presenter.SetWeaponId("arcade.weapon.basic"))throw new Exception(presenter.LastError);
            presenter.SetManualPoseSampling(true);presenter.BodyAnimator.Rebind();presenter.BodyAnimator.Update(0);
            presenter.Current.Animations.Idle.Clip.SampleAnimation(presenter.BodyAnimator.gameObject,0);presenter.ApplyOffhandGrip(presenter.Current.Animations.Idle);
            foreach(var animator in player.GetComponentsInChildren<Animator>())animator.enabled=false;
            foreach(var canvas in player.GetComponentsInChildren<Canvas>(true))canvas.gameObject.SetActive(false);
            foreach(var particle in player.GetComponentsInChildren<ParticleSystem>(true))particle.Stop(true,ParticleSystemStopBehavior.StopEmittingAndClear);
            var p=catalog.Arenas.Single(a=>a.ArenaNumber==2).Hazards.First(h=>h.Kind==1);
            var hazard=new[]{new ArcadeHazardParameters(p.Id,p.Kind,4.6f,0,p.DirectionDegrees,p.Radius,p.ArmLength,p.ArmHalfWidth,p.IdleSeconds,p.WarningSeconds,p.ActiveSeconds,p.AngularSpeedDegrees,p.HitIntervalSeconds,p.PlayerDamageRatio,p.BossDamageRatio)};
            var camera=new GameObject("StudyCamera",typeof(Camera)).GetComponent<Camera>();camera.cullingMask=1<<30;camera.orthographic=true;camera.orthographicSize=3.5f;
            var focus=new Vector3(0,1,0);camera.transform.position=focus+new Vector3(0,8,-14);camera.transform.LookAt(focus);
            camera.clearFlags=CameraClearFlags.SolidColor;camera.backgroundColor=new Color(.08f,.11f,.15f);
            var light=new GameObject("Key",typeof(Light)).GetComponent<Light>();light.type=LightType.Directional;light.intensity=1.5f;light.transform.eulerAngles=new Vector3(45,-35,0);light.cullingMask=1<<30;
            var floor=GameObject.CreatePrimitive(PrimitiveType.Cube);floor.transform.position=new Vector3(0,-.08f,0);floor.transform.localScale=new Vector3(40,.1f,40);
            floor.GetComponent<Renderer>().sharedMaterial=AssetDatabase.LoadAssetAtPath<Material>("Assets/Game/Art/Arcade/Environment/Content/Slate.mat");
            target=new RenderTexture(1600,900,24);image=new Texture2D(1600,900,TextureFormat.RGB24,false);camera.targetTexture=target;
            string folder=Path.GetFullPath("../Docs/Plans/Artifacts/BarrelTowerVisualStudy-20261020");Directory.CreateDirectory(folder);
            double[] times={0,4.1,5.6};string[] names={"current-safe","current-warning","current-active"};
            for(int index=0;index<times.Length;index++)
            {
                view.SetRoundContent(times[index],new[]{new ArcadeContentAreaSnapshot(-10002,13,1.4f,0,.45f,false)});view.SetHazards(hazard,times[index]);
                foreach(var item in scene.GetRootGameObjects())foreach(var t in item.GetComponentsInChildren<Transform>(true))t.gameObject.layer=30;
                Save(camera,target,image,Path.Combine(folder,names[index]+".png"));
            }
            camera.transform.position=focus+new Vector3(0,16,-11);camera.transform.LookAt(focus);camera.orthographicSize=4.15f;
            Save(camera,target,image,Path.Combine(folder,"current-topdown.png"));
            var body=player.GetComponentsInChildren<SkinnedMeshRenderer>().FirstOrDefault(r=>r.enabled&&r.gameObject.activeInHierarchy);
            if(body==null)throw new Exception("正式主角未显示");
            File.WriteAllText(Path.Combine(folder,"capture.txt"),"Source: production prefab and view projection; isolated scene; no resource edits.\nHero: "+body.name+"; bounds="+body.bounds.size+"\nTimes: 0,4.1,5.6\n");
            camera.targetTexture=null;return folder;
        }
        finally
        {
            if(view!=null){var flags=BindingFlags.Instance|BindingFlags.NonPublic;((IDisposable)typeof(ArcadeCombatViewController).GetField("hazardPresentation",flags).GetValue(view))?.Dispose();typeof(ArcadeCombatViewController).GetMethod("ReleaseRoundContent",flags).Invoke(view,null);}
            RenderTexture.active=oldTarget;RenderSettings.ambientMode=mode;RenderSettings.ambientLight=ambient;
            if(image!=null)Object.DestroyImmediate(image);if(target!=null){target.Release();Object.DestroyImmediate(target);}
            SceneManager.SetActiveScene(prior);EditorSceneManager.CloseScene(scene,true);
        }
    }
    private static void Save(Camera camera,RenderTexture target,Texture2D image,string path)
    {camera.Render();RenderTexture.active=target;image.ReadPixels(new Rect(0,0,target.width,target.height),0,0);image.Apply();File.WriteAllBytes(path,image.EncodeToPNG());}
}
